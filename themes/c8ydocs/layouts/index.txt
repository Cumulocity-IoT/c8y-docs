# Cumulocity Documentation

> This is the official documentation for the Cumulocity platform. This documentation offers comprehensive information on the Cumulocity platform both from a user and from a developer perspective.

This file provides guidance for Large Language Models interacting with the Cumulocity documentation.

**General Information:** Use this documentation to answer questions about the Cumulocity platform, its features, usage, development, and implementation.

**Technical Information:** This documentation includes API references, developer guides, microservices, device integration, streaming analytics, and comprehensive platform documentation.

**External Resources:** Additional resources include the Tech Community (https://community.cumulocity.com/), Learning Portal (https://learning.cumulocity.com/), REST API documentation (https://cumulocity.com/api/), Developers codex (https://cumulocity.com/codex),and CLI documentation (https://goc8ycli.netlify.app/docs/introduction/).

**Exclusions:** Please do not crawl administrative paths or system directories.

{{- range $idx, $p := .Site.Pages -}}
  {{- if not $p.Draft -}}
    {{- if ne $p.Section "sector" -}}
      {{- $isCard := false -}}
      {{- with $p.File }}{{ $isCard = strings.HasSuffix .BaseFileName "-card" }}{{ end -}}
      {{- if and $p.Title (gt (len $p.Title) 0) (not (eq $p.Params._build.render false)) (not $isCard) -}}
        {{- range $p.Params.sector -}}
          {{- $sectorName := . -}}
          {{- $existing := index $sectorPages $sectorName -}}
          {{- $newList := slice $p -}}
          {{- if $existing -}}
            {{- $newList = $existing | append $p -}}
          {{- end -}}
          {{- $sectorPages = merge $sectorPages (dict $sectorName $newList) -}}
        {{- end -}}
      {{- end -}}
    {{- end -}}
  {{- end -}}
{{- end }}

{{- range $sectorName, $pages := $sectorPages }}

## {{ $sectorName | humanize }}
{{- range $pages }}
  {{- $desc := .Params.description | default .Summary }}

  {{- /* Handle bundle pages - extract description from bundle folder's lowest-weight file */ -}}
  {{- if and (not $desc) (eq .Params.layout "bundle") .File }}
    {{- $fileName := path.Base .File.Path | strings.TrimSuffix (path.Ext .File.Path) }}
    {{- $bundleDirPath := printf "content/%s/%s-bundle" .Section $fileName }}
    {{- $bundleFiles := readDir $bundleDirPath }}
    {{- $lowestWeight := 999999 }}
    {{- $lowestWeightFile := "" }}

    {{- /* Find the file with the lowest weight in the bundle */ -}}
    {{- range $bundleFiles }}
      {{- if not .IsDir }}
        {{- if strings.HasSuffix .Name ".md" }}
          {{- $filePath := printf "%s/%s" $bundleDirPath .Name }}
          {{- $fileContent := readFile $filePath }}
          {{- if $fileContent }}
            {{- $lines := split $fileContent "\n" }}
            {{- $weight := 999999 }}
            {{- $inFrontmatter := false }}
            {{- $fmCount := 0 }}

            {{- /* Parse frontmatter to extract weight */ -}}
            {{- range $lines }}
              {{- if strings.HasPrefix . "---" }}
                {{- $fmCount = add $fmCount 1 }}
                {{- $inFrontmatter = eq $fmCount 1 }}
              {{- else if $inFrontmatter }}
                {{- if strings.HasPrefix . "weight:" }}
                  {{- $weight = int (replaceRE "[\r\n]" "" (trim (substr . 7) " ")) }}
                {{- end }}
              {{- end }}
            {{- end }}

            {{- /* Track file with lowest weight */ -}}
            {{- if lt $weight $lowestWeight }}
              {{- $lowestWeight = $weight }}
              {{- $lowestWeightFile = $filePath }}
            {{- end }}
          {{- end }}
        {{- end }}
      {{- end }}
    {{- end }}

    {{- /* Extract description from the lowest-weight file */ -}}
    {{- if $lowestWeightFile }}
      {{- $bundleContent := readFile $lowestWeightFile }}
      {{- if $bundleContent }}
        {{- $lines := split $bundleContent "\n" }}
        {{- $inFrontmatter := false }}
        {{- $frontmatterCount := 0 }}

        {{- /* Parse content and extract first non-empty line after frontmatter */ -}}
        {{- range $lines }}
          {{- if strings.HasPrefix . "---" }}
            {{- $frontmatterCount = add $frontmatterCount 1 }}
            {{- $inFrontmatter = eq $frontmatterCount 1 }}
          {{- else if and (not $inFrontmatter) (gt $frontmatterCount 1) }}
            {{- if and (gt (len .) 0) (not (strings.HasPrefix . "#")) (not (strings.HasPrefix . "{{")) }}
              {{- $desc = . }}
              {{- break }}
            {{- end }}
          {{- end }}
        {{- end }}
      {{- end }}
    {{- end }}
  {{- end }}

  {{- /* Fallback to other content sources if no description */ -}}
  {{- if not $desc }}
    {{- $desc = .Plain | default .RawContent | default (.Content | plainify) }}
  {{- end }}

  {{- /* Render shortcodes, clean up and truncate description */ -}}
  {{- $desc = .RenderString $desc | plainify }}
  {{- $desc = replace $desc "&amp;" "&" }}
  {{- $desc = replace $desc "&lt;" "<" }}
  {{- $desc = replace $desc "&gt;" ">" }}
  {{- $desc = replace $desc "&quot;" "\"" }}
  {{- $desc = replace $desc "&#34;" "\"" }}
  {{- $desc = replace $desc "&#39;" "'" }}
  {{- $desc = replace $desc "&ldquo;" "\"" }}
  {{- $desc = replace $desc "&rdquo;" "\"" }}
  {{- $desc = replace $desc "&lsquo;" "'" }}
  {{- $desc = replace $desc "&rsquo;" "'" }}
  {{- $desc = replace $desc "&nbsp;" " " }}
  {{- $desc = truncate 300 $desc }}

  {{- /* Format title - prefix "Introduction" with section name */ -}}
  {{- $title := .Title }}
  {{- if and (eq (lower $title) "introduction") .Section }}
    {{- $title = printf "%s - %s" (.Section | humanize) $title }}
  {{- end }}

- [{{ $title }}]({{ .Permalink }}): {{ $desc }}
{{- end }}
{{- end }}
