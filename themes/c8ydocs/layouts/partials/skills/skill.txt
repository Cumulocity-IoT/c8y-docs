{{- /*
  Returns a skill in agent skill format (SKILL.md): front matter with `name`, `description`,
  and `compatibility`, then the skill as in llms-full.txt with the skills it builds on inserted
  after the metadata lines, then the API operations it uses and the documentation that links
  to it. The skill's index.md and the skills download both use it. Call with the skill page.
*/ -}}
{{- $page := . -}}
{{- $usedIn := index (partialCached "skills/index.html" site) .File.ContentBaseName -}}

{{- /* llms-page.txt returns "# Title", the metadata lines and the body, separated by blank lines */ -}}
{{- $parts := split (partial "llms-page.txt" .) "\n\n" -}}
{{- $sections := slice (index $parts 0) (index $parts 1) -}}
{{- with partial "skills/requires.html" . -}}
  {{- $lines := slice -}}
  {{- range . -}}
    {{- $lines = $lines | append (printf "- `%s` skill: [%s](%s)" (partial "skills/skill-name.html" .) .Title (.OutputFormats.Get "markdown").Permalink) -}}
  {{- end -}}
  {{- $sections = $sections | append (printf "## Before you start\n\nUse these skills first:\n\n%s" (delimit $lines "\n")) -}}
{{- end -}}
{{- $sections = $sections | append (strings.Trim (delimit (after 2 $parts) "\n\n") "\n") -}}

{{- with partial "skills/api-groups.html" . -}}
  {{- $groups := slice -}}
  {{- range . -}}
    {{- $lines := slice -}}
    {{- range .entries }}{{ $lines = $lines | append (printf "- [%s %s](%s): %s" .method .path .url .title) }}{{ end -}}
    {{- $groups = $groups | append (printf "### %s\n\nReference: %s\n\n%s" .title .root (delimit $lines "\n")) -}}
  {{- end -}}
  {{- $sections = $sections | append (printf "## API reference\n\n%s" (delimit $groups "\n\n")) -}}
{{- end -}}

{{- with $usedIn -}}
  {{- $lines := slice -}}
  {{- range . -}}
    {{- $title := replace .page.Title "var-product-c8y-iot" site.Params.product_c8y_iot -}}
    {{- with .from }}{{ $title = printf "%s > %s" $title (replace . "var-product-c8y-iot" site.Params.product_c8y_iot) }}{{ end -}}
    {{- $lines = $lines | append (printf "- [%s](%s%s)" $title .page.Permalink (cond (ne .anchor "") (printf "#%s" .anchor) "")) -}}
  {{- end -}}
  {{- $sections = $sections | append (printf "## Related documentation\n\n%s" (delimit $lines "\n")) -}}
{{- end -}}

{{- /* Agents pick a skill from its description alone, so the Agent Skills specification asks for
  what the skill does and when to use it: the page description, then the `when` front matter.
  `compatibility` (optional in the specification) names what the requests need. */ -}}
{{- $description := .Description -}}
{{- with .Params.when }}{{ $description = printf "%s Use when %s." $description . }}{{ end -}}
{{- $compatibility := "Requires network access to a Cumulocity tenant and the C8Y_BASEURL, C8Y_TENANT, C8Y_USER, and C8Y_PASSWORD environment variables." -}}
{{- $frontMatter := printf "---\nname: %s\ndescription: %s\ncompatibility: %s\n---" (partial "skills/skill-name.html" .) ($description | jsonify) ($compatibility | jsonify) -}}
{{- return printf "%s\n\n%s\n" $frontMatter (delimit $sections "\n\n") -}}
