{{- /* Returns the recipes a documentation page links to with its `recipes` front matter (a list of file names in content/recipes/), as a slice of dicts: - "recipe": the recipe page - "from": title of the bundle block that names the recipe, empty for the page itself - "anchor": id of that block (its file name, as _default/bundle.html renders it) Bundle pages also collect the `recipes` of the blocks in their headless -bundle folder, in weight order. Each recipe is returned once. A name that matches no recipe fails the build. Call with the page. */ -}} {{- $refs := slice -}} {{- $path := "" }}{{ with .File }}{{ $path = .Path }}{{ end -}} {{- range .Params.recipes }}{{ $refs = $refs | append (dict "slug" . "from" "" "anchor" "" "file" $path) }}{{ end -}} {{- with .File -}} {{- with site.GetPage (printf "%s-bundle" .BaseFileName) -}} {{- range sort (.Resources.ByType "page") ".Params.weight" -}} {{- $block := . -}} {{- range .Params.recipes -}} {{- $refs = $refs | append (dict "slug" . "from" $block.Title "anchor" (substr $block.Name 0 -3) "file" $block.File.Path) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $seen := slice -}} {{- $result := slice -}} {{- range $refs -}} {{- if in $seen .slug }}{{ continue }}{{ end -}} {{- $seen = $seen | append .slug -}} {{- $recipe := site.GetPage (printf "/recipes/%s" .slug) -}} {{- if not $recipe -}} {{- errorf "recipe %q referenced by %s does not exist in content/recipes/" .slug .file -}} {{- continue -}} {{- end -}} {{- $result = $result | append (dict "recipe" $recipe "from" .from "anchor" .anchor) -}} {{- end -}} {{- return $result -}}
