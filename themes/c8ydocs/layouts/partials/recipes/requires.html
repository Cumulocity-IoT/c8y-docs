{{- /* Returns the recipes that a recipe builds on (its `requires` front matter, a list of file names in content/recipes/) as a slice of pages. A name that matches no recipe fails the build. Call with the recipe page. */ -}} {{- $page := . -}} {{- $result := slice -}} {{- range .Params.requires -}} {{- with site.GetPage (printf "/recipes/%s" .) -}} {{- $result = $result | append . -}} {{- else -}} {{- errorf "recipe %q required by %s does not exist in content/recipes/" . $page.File.Path -}} {{- end -}} {{- end -}} {{- return $result -}}
